rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() { return signedIn() && request.auth.token.email == "jbonilla2987@gmail.com"; }
    function own(uid) { return signedIn() && request.auth.uid == uid; }
    function approved(uid) {
      return isAdmin() || (
        own(uid) &&
        exists(/databases/$(database)/documents/accessRequests/$(uid)) &&
        get(/databases/$(database)/documents/accessRequests/$(uid)).data.status == "approved"
      );
    }

    match /accessRequests/{uid} {
      allow list: if isAdmin();
      allow get: if isAdmin() || own(uid);
      allow create: if isAdmin() || (
        own(uid) &&
        request.resource.data.status == "pending" &&
        request.resource.data.role == "user" &&
        request.resource.data.email == request.auth.token.email
      );
      allow update, delete: if isAdmin();
    }

    match /users/{uid} {
      allow read: if isAdmin() || approved(uid);
      allow create, update, delete: if approved(uid) && own(uid);
      match /{document=**} {
        allow read: if isAdmin() || approved(uid);
        allow create, update, delete: if approved(uid) && own(uid);
      }
    }

    match /liveLocations/{uid} {
      allow read: if isAdmin();
      allow create, update: if approved(uid) && own(uid)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email;
      allow delete: if isAdmin() || (approved(uid) && own(uid));
    }

    match /dataExportRequests/{uid} {
      allow list: if isAdmin();
      allow read: if isAdmin() || own(uid);
      allow create: if approved(uid) && own(uid)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == "pending";
      allow update: if isAdmin() || (approved(uid) && own(uid));
      allow delete: if isAdmin();
    }

    match /dataDeletionRequests/{uid} {
      allow list: if isAdmin();
      allow read: if isAdmin() || own(uid);
      allow create: if approved(uid) && own(uid)
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == "pending";
      allow update: if isAdmin() || (approved(uid) && own(uid) && request.resource.data.status == "pending");
      allow delete: if isAdmin();
    }

    match /supportThreads/{uid} {
      allow list: if isAdmin();
      allow read, create, update: if isAdmin() || (approved(uid) && own(uid));
      allow delete: if isAdmin();
      match /messages/{messageId} {
        allow read: if isAdmin() || (approved(uid) && own(uid));
        allow create: if isAdmin() || (
          approved(uid) && own(uid) && request.resource.data.senderUid == request.auth.uid
        );
        allow update, delete: if isAdmin();
      }
    }

    match /chatDirectory/{uid} {
      allow read: if signedIn();
      allow create, update: if approved(uid) && own(uid)
        && request.resource.data.uid == request.auth.uid;
      allow delete: if isAdmin() || (approved(uid) && own(uid));
    }

    match /userChats/{chatId} {
      allow read: if signedIn() && request.auth.uid in resource.data.participants;
      allow create: if signedIn()
        && request.resource.data.participants.size() == 2
        && request.auth.uid in request.resource.data.participants;
      allow update: if signedIn()
        && request.auth.uid in resource.data.participants
        && request.resource.data.participants == resource.data.participants;
      allow delete: if isAdmin();

      match /messages/{messageId} {
        function parentChat() { return get(/databases/$(database)/documents/userChats/$(chatId)); }
        allow read: if signedIn() && request.auth.uid in parentChat().data.participants;
        allow create: if signedIn()
          && request.auth.uid in parentChat().data.participants
          && request.resource.data.senderUid == request.auth.uid;
        allow update, delete: if isAdmin();
      }
    }
  }
}
